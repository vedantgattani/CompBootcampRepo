wc -l ../data/fasta/407228326.fasta ../data/fasta/407228412.fasta ../data/fasta/e_coli.fasta

# As mentioned in assignment, the final lines of the two fasta files are not new-line terminates, 
# so wc -l counts one less line than there acually is (since wc -l only counts newline characters)

#2
tail -n +2 ../data/fasta/e_coli.fasta

# The above code prints everything from the second line in e_coli.fasta till the end. Since the header is the first line, it is not printed.
# the tail command is meant to print the last ten lines of a file by default, but -n +2  makes it start displaying from the second line of 
# the file, and continue till the end.

#3
tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\n' | wc -c

# The first part removes the header and the second (middle) part deletes newline characters and joins all lines into a single string.
# The third command, similar to wc -l, counts all letters (symbols), including ambiguous nucleotides such as N, R, Y, M, and W.
## I found tr -d '\n' on stack overflow, and used it to remove newline characters - also later found in class notes ##

#4
tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\n' | grep -o 'ATGC' | wc -l

# The first two commands remain the same as before, and are used to remove the header and newline characters. grep -o, as mentioned
# in the assignment, outputs each individual occurance of A, T, G, and C, and wc -l counts those occurances.

#5
echo "scale = 5 ; $(tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\n' | grep -o '[AT]' | wc -l) / $(tail -n +2 ../data/fasta/e_coli.fasta | tr -d '\n' | grep -o '[GC]' | wc -l)" | bc -l

# As bc is a command run by bash (not part of bash), it needs echo since the output is directly sent to the bc command at the end. 
# scale = 5 tells bc how many decimel places to use - I used five, since original answer was many decimel places. 
# $() is used to run the commands in the brackets first, as given in notes. So, essentially $(A+T) and $(G+C).
# Chunks of commands in both brackets do the same as in the last answer, except for only AT and GC respectively.
# This includes getting rid of header and newline characters, finding every occurance of ATs and GCs and then wc -l counting them.
# Outputs from both those brackets are divided using "/" to find the ratio. 
# Eventually, the output is then sent to bc -l (using echo in the beginning), to show the result.
## I found out how to correctly use bc -l (by using echo) and what scale = x does on stack overflow ##
